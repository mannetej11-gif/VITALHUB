export default function UnifiedApp() {
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [user, setUser] = useState(null);
  const [token, setToken] = useState(localStorage.getItem("jwt_token"));
  const [activeModule, setActiveModule] = useState("dashboard");
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    if (token) verifyToken();
  }, [token]);

  const verifyToken = async () => {
    try {
      const response = await fetch(`${API_URL}/auth/profile`, {
        headers: { "Authorization": `Bearer ${token}` }
      });
      if (response.ok) {
        const userData = await response.json();
        setUser(userData);
        setIsAuthenticated(true);
      } else {
        logout();
      }
    } catch {
      logout();
    }
  };

  const handleLogin = async (username, password) => {
    setLoading(true);
    try {
      const response = await fetch(`${API_URL}/auth/login`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ username, password })
      });
      const data = await response.json();
      if (response.ok) {
        localStorage.setItem("jwt_token", data.access_token);
        setToken(data.access_token);
        setUser(data.user);
        setIsAuthenticated(true);
      } else {
        alert("Login failed: " + data.error);
      }
    } catch (err) {
      alert("Connection error: " + err.message);
    }
    setLoading(false);
  };

  const handleRegister = async (username, email, password, fullName) => {
    setLoading(true);
    try {
      const response = await fetch(`${API_URL}/auth/register`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ username, email, password, full_name: fullName })
      });
      const data = await response.json();
      if (response.ok) {
        alert("Registration successful! Please login.");
      } else {
        alert("Registration failed: " + data.error);
      }
    } catch (err) {
      alert("Connection error: " + err.message);
    }
    setLoading(false);
  };

  const logout = () => {
    localStorage.removeItem("jwt_token");
    setToken(null);
    setUser(null);
    setIsAuthenticated(false);
    setActiveModule("dashboard");
  };

  if (!isAuthenticated) {
    return <LoginPage onLogin={handleLogin} onRegister={handleRegister} loading={loading} />;
  }

  return <MainApp user={user} activeModule={activeModule} setActiveModule={setActiveModule} onLogout={logout} />;
}

      </div>
    </div>
  );
}

function PriorityBadge({ priority }) {
  const colors = {
    Critical: { bg: "#FEE2E2", fg: "#991B1B" },
    High: { bg: "#FEF3C7", fg: "#B45309" },
    Moderate: { bg: "#DBEAFE", fg: "#0284C7" },
    Routine: { bg: "#DCFCE7", fg: "#15803D" }
  };
  const c = colors[priority] || colors.Routine;
  return <span style={{ fontSize: "11px", padding: "6px 12px", borderRadius: "6px", background: c.bg, color: c.fg, fontWeight: 600 }}>{priority}</span>;
}

function StatusBadge({ status }) {
  const colors = { contacted: { bg: "#E0F2FE", fg: "#0284C7" }, meeting_scheduled: { bg: "#FEF3C7", fg: "#B45309" }, proposal_sent: { bg: "#D1FAE5", fg: "#047857" } };
  const c = colors[status] || colors.contacted;
  return <span style={{ fontSize: "11px", padding: "6px 12px", borderRadius: "6px", background: c.bg, color: c.fg, fontWeight: 600 }}>{status.replace(/_/g, " ").toUpperCase()}</span>;
}

function SentimentBadge({ sentiment }) {
  const colors = { positive: { bg: "#D1FAE5", fg: "#047857" }, neutral: { bg: "#E5E7EB", fg: "#4B5563" }, negative: { bg: "#FEE2E2", fg: "#991B1B" } };
  const c = colors[sentiment] || colors.neutral;
  return <span style={{ fontSize: "11px", padding: "6px 12px", borderRadius: "6px", background: c.bg, color: c.fg, fontWeight: 600 }}>{sentiment.toUpperCase()}</span>;
}
